## Contract

### Pattern catalogue

Four patterns cover the runtime rules of TADR-002 as refined by HF-04a. Each names the owning HF-03a module, the interfaces already listed in HF-03a's table, and the uses it excludes. A pattern is a way of arranging owned records and interfaces that already exist in HF-03a; it adds no module, import edge or global.

Encounter is the only gameplay-time owner (HF-04a line 20). The four patterns order themselves around its tick as follows. Only the first row advances time.

```gdscript
# unparsed: no editor run in this card; scene replacement and scoped teardown await a later authorized proof
extends Node

var control: SessionControl
var run_state: RunState
var pending_transition: PendingTransition
var encounter: EncounterCoordinator
var presentation: Presentation
var input: InputAdapter
var content: ContentAdmission
var scoped_connections: Array[ScopedConnection] = []


func replace_encounter(request: ReplaceEncounterRequest) -> TransitionStatus:
	var outgoing: EncounterIdentity = control.active_encounter
	control.closing_encounter = outgoing
	input.clear_physical_input()
	encounter.cancel_prepared()
	request_pause()
	var results: Array[EncounterResult] = encounter.read_committed_results()
	for result: EncounterResult in results:
		accept_encounter_result(result)
	if pending_transition.blocks_replacement():
		return TransitionStatus.new(outgoing, TransitionStatus.HELD, pending_transition.diagnostic)
	encounter.retire()
	for connection: ScopedConnection in scoped_connections:
		connection.release()
	scoped_connections.clear()
	presentation.release_encounter_views()
	_free_outgoing_composition(encounter)
	var incoming: EncounterIdentity = EncounterIdentity.next(outgoing)
	var handle: CatalogueHandle = content.retain_revision(run_state.content_revision)
	var build: EncounterBuildData = _encounter_values(handle, run_state, request)
	encounter = _compose_encounter(incoming, build)

		return TransitionStatus.new(incoming, TransitionStatus.HELD, readiness.diagnostic)
	control.active_encounter = incoming
	control.closing_encounter = EncounterIdentity.none()
	presentation.present_snapshot(encounter.read_committed_results())
	_bind_scoped_results(incoming)
	return resume()


func _bind_scoped_results(identity: EncounterIdentity) -> void:
	var connection: ScopedConnection = ScopedConnection.new(identity, encounter.committed_result_available, _on_committed_result_available)
	connection.establish()
	scoped_connections.append(connection)


func _on_committed_result_available(identity: EncounterIdentity, result_id: ResultIdentity) -> void:
	if identity != control.active_encounter or identity == control.closing_encounter:
		return
	control.pending_result_reads.append(result_id)
```
