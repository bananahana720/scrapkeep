extends SceneTree
# Render-bench throwaway (not in the repo): the class-select screen and the customize page for
# Tally and Cannon, at 1920x1080, driven the way tests/ui_customize_page_test.gd drives them.
#   godot --path <core-loop> --resolution 1920x1080 --script <this> -- --out=<dir>
const Game := preload("res://src/game.gd")
const SIZE := Vector2i(1920, 1080)
var out := ""
var mode := "customize"
var _game


func _initialize() -> void:
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--out="):
			out = a.trim_prefix("--out=")
		elif a.begins_with("--mode="):
			mode = a.trim_prefix("--mode=")
	if DisplayServer.get_name() == "headless" or out == "":
		printerr("needs a display and --out")
		quit(2)
		return
	DirAccess.make_dir_recursive_absolute(out)
	root.size = SIZE
	root.content_scale_size = SIZE
	root.content_scale_mode = Window.CONTENT_SCALE_MODE_VIEWPORT
	_game = Game.new()
	_game.test_mode = true
	root.add_child(_game)
	_run.call_deferred()


func _frames(n := 2) -> void:
	for i in n:
		await process_frame


func _snap(name: String) -> void:
	await _frames(3)
	var img: Image = root.get_texture().get_image()
	img.save_png(out.path_join(name + ".png"))
	print("saved ", name, " ", img.get_size())


func _run() -> void:
	while _game.flow == null:
		await process_frame
	await _frames(5)
	_game.ctx.calm = true
	var profile = _game.ctx.profile
	profile.unlock_pilot("tally")
	profile.unlock_pilot("remembered_cannon")
	if mode == "class":
		# Class select (the class cards), as screens.gd draws it on the "class_select" state.
		_game.flow.press_start()
		await _frames(30)
		await _snap("class-select")
		quit(0)
		return
	# Customize page, via the camp's lockers station (the test's route).
	var screens = _game.screens
	_game._screen_key(KEY_K)
	await _frames()
	var hub = _game.camp_hub
	var toward: Vector3 = hub.root.global_position - hub.zone_position("lockers")
	toward.y = 0.0
	var at: Vector3 = hub.zone_position("lockers") + toward.normalized() * 1.5
	at.y = 1.3
	_game.player.reset_to(at)
	_game.player.body().global_position = at
	for i in 4:
		await physics_frame
	_game._screen_key(KEY_C)
	await _frames()
	var camp = screens.camp_page()
	var page = camp.get_node_or_null("CustomizePage")
	if page == null or not page.is_visible_in_tree():
		printerr("customize page did not open")
		quit(3)
		return
	page.reduced_motion = true
	for id in ["tally", "remembered_cannon"]:
		page.pick_pilot(id)
		await _frames(40)
		await _snap("customize-" + id)
	var f := FileAccess.open(out.path_join("menu_capture.json"), FileAccess.WRITE)
	f.store_string(JSON.stringify({"adapter": RenderingServer.get_video_adapter_name(), "pilots": page._pilots}, "\t"))
	f.close()
	quit(0)
